\section{Discussion \& Implications}

\begin{frame}
\frametitle{Why Silicon Valley? Geographic Clustering Effects}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Silicon Valley advantages:}
\begin{itemize}
    \item Dense information networks
    \item Proximity to talent \& infrastructure
    \item Shared risk assessment practices
    \item Established trust relationships
\end{itemize}
\end{column}

\begin{column}{0.5\textwidth}
\textbf{Comparison with Comm. 0:}
\begin{itemize}
    \item Similar size \& U.S. focus
    \item But \textbf{75\% lower} transaction volume
    \item Non-nested organization
    \item Less efficient capital allocation
\end{itemize}
\end{column}
\end{columns}

\vspace{0.3cm}
\begin{alertblock}{Key Insight}
Network \textbf{topology}, not just size or geography, determines ecosystem efficiency.
\end{alertblock}

\end{frame}

\begin{frame}
\frametitle{Robustness Trade-offs: Stability vs. Vulnerability}

\textbf{Ecological Analogy:}
\begin{itemize}
    \item Nested networks are \textcolor{green}{\textbf{resilient}} to random failures
    \item But \textcolor{red}{\textbf{vulnerable}} to targeted hub removal
    \item "Mutualistic trade-offs" in ecosystem stability
\end{itemize}

\vspace{0.3cm}

\textbf{VC Implications:}
\begin{itemize}
    \item Stable against random investor departures
    \item Fragile to key hub investor exits
    \item Concentration risk in Silicon Valley
    % \item Need for diversification strategies
\end{itemize}
\end{frame}
