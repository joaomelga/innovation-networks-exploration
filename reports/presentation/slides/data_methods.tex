\textbf{Geographic \& Temporal Scope:}

    \item Minimum funding threshold: \$150,000
\end{itemize}

\begin{frame}
\frametitle{Network Construction Process}

\begin{enumerate}
    \item \textbf{Stage Classification:}
    \begin{itemize}
        \item Early stages: Angel, Pre-seed, Seed, Series A
        \item Late stages: Series B through Series I
    \end{itemize}
    \vspace{0.3cm}

    \item \textbf{Bipartite Network Creation:}
    \begin{itemize}
        \item Two disjoint investor sets: Early vs. Late
        \item Edges = co-investment in same companies
        \item 169,679 investment pairs across 3,666 startups
    \end{itemize}
    \vspace{0.3cm}
    
    \item \textbf{Community Detection:}
    \begin{itemize}
        \item Greedy modularity optimization
        \item Identifies ~168 communities
        \item Focus on 7 communities with 150+ investors
    \end{itemize}
\end{enumerate}

\end{frame}
